import { css } from '@emotion/react';

import { useMedia, Button } from '@wanteddev/wds';

const Component = () => {
const { breakpoint, respondTo } = useMedia();

    return (
    	<Button
    		variant="outlined"
    		color="primary"
    		css={{
    			[respondTo(breakpoint.md)]: {
    				marginTop: '5px',
    			},
    		}}
    	>
    		Button
    	</Button>
    );

};
