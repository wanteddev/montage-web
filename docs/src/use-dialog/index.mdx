# Dialog

<Primary />

기본적으로 `useDialog` 를 활용하여 호출합니다.

dialog는 `Promise<'confirm' | 'cancel'>` 타입을 반환합니다.

<Markdown>
{`
\`\`\`tsx
import { useDialog } from '@wanteddev/wds';

const Component = () => {
const dialog = useDialog();

const handleClick = async () => {
const result = await dialog({
title: '제목',
content: '설명',
confirmText: '확인',
cancelText: '취소',
});

    alert('결과: ' + result);

};

return (
<button onClick={handleClick}>
Click Me!
</button>
);
};
\`\`\`
`}

</Markdown>
