# useMedia

Theme 객체에서 직접 꺼내어 여러 미디어 쿼리를 다룹니다.

```tsx
import { Box, useMedia } from '@wanteddev/wds';

const Component = () => {
  const { respondTo, respondMore, breakpoint } = useMedia();

  return (
    <Box
      sx={{
        width: '100px',
        [respondTo(breakpoint.xs)]: {
          width: '200px',
        },
      }}
    />
  );
};
```
