# useSnackbar

스낵바는 프로세스에 대한 간단한 메시지를 제공하며 이를 제어 가능합니다.

`RegionConfig` 컴포넌트를 활용하여 노출되는 위치를 조정할 수 있습니다.

## 활용

- 사용자가 한 행동에 대한 피드백을 화면 하단에 보여줍니다.
- 프로세스에 대한 간단한 메시지와 별도의 버튼을 제공합니다.
- 버튼으로 사용자와 상호작용할 수 있습니다.
- \[X\] 아이콘 버튼은 필요에 따라 노출하며, 버튼을 누르면 화면에서 즉시 사라집니다.
- 가독성을 위해 메시지가 2줄 이상 넘지 않도록 작성합니다.

<Demo code={`import { useSnackbar, Button } from '@wanteddev/wds';

const Demo = () => {
  const snackbar = useSnackbar();

  const handleClick = () => {
    snackbar({
      variant: 'normal',
      heading: '메시지에 마침표를 찍어요.',
      action: {
        children: '텍스트'
      }
    });
  };

  return (
    <Button onClick={handleClick}>클릭</Button>
  )
}

export default Demo;`}
/>

## Props

<PropsTable component="useSnackbar" fallback={[
{ name: 'duration', types: 'number | undefined', defaultValue: 5000 },
{ name: 'variant', types: '\"normal\" | undefined', defaultValue: 'normal' },
{ name: 'heading', types: 'ReactNode' },
{ name: 'description', types: 'ReactNode' },
{ name: 'extraContent', types: 'ReactNode' },
{ name: 'action', types: 'ComponentProps<typeof TextButton>', required: true },
{ name: 'onAnimationEnd', types: '(type: \"hide\" | \"show\") => void' }
]} />

## Example

<Demo code={`import { FlexBox, useSnackbar, Button, Avatar } from '@wanteddev/wds';
import { IconCircleFill } from '@wanteddev/wds-icon';

const description = '설명이 두 줄 이상 들어갈 때 예외적으로 쓸 수 있게끔 고려해두었습니다.';
const heading = '메시지에 마침표를 찍어요.';
const extraContent = <Avatar variant="company" size="medium" />

const Demo = () => {
  const snackbar = useSnackbar();

  const handleClick = (options = {}) => () => {
    snackbar({ action: { children: '텍스트' },...options });
  };

  return (
    <FlexBox gap="8px">
      <Button onClick={handleClick({ heading })}>heading</Button>
      <Button onClick={handleClick({ description })}>description</Button>
      <Button onClick={handleClick({ heading, description })}>heading + description</Button>
      <Button onClick={handleClick({ heading, extraContent })}>extraContent1</Button>
      <Button onClick={handleClick({ heading, description, extraContent })}>extraContent2</Button>
    </FlexBox>
  )
}

export default Demo;`}
/>
