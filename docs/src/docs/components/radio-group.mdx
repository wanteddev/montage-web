# RadioGroup

사용자에게 입력해야 할 정보를 명확하게 인지시키고, 정보를 정확하게 입력할 수 있도록 도와줍니다.

## 활용

<Demo code={`import { useId } from 'react';
import { FlexBox, RadioGroup, Label, RadioGroupItem } from '@wanteddev/wds';

const Demo = () => {
  const manId = useId();
  const womanId = useId();

  return (
    <RadioGroup>
      <FlexBox gap="8px" alignItems="center">
        <RadioGroupItem id={manId} value="man" />
        <Label htmlFor={manId}>Man</Label>
      </FlexBox>

      <FlexBox gap="8px" alignItems="center">
        <RadioGroupItem id={womanId} value="woman" />
        <Label htmlFor={womanId}>Woman</Label>
      </FlexBox>
    </RadioGroup>
  )
}

export default Demo;`}
/>

## Example

### Form

<Demo code={`import { Button, FlexBox, RadioGroup, RadioGroupItem, FormField, FormControl, FormLabel, FormErrorMessage } from '@wanteddev/wds';
import { useForm, Controller } from 'react-hook-form';

const Demo = () => {
  const form = useForm();

  return (
    <FlexBox
      as="form"
      flexDirection="column"
      gap="20px"
      onSubmit={form.handleSubmit((v) => alert(JSON.stringify(v)))}
    >
      <Controller
        control={form.control}
        name="selected"
        rules={{
          required: {
            value: true,
            message: "필수 값입니다.",
          },
        }}
        render={({ field, formState }) => (
          <FormField>
            <FormLabel>성별</FormLabel>

            <FormControl>
              <RadioGroup {...field} onValueChange={field.onChange}>
                <FormField flexDirection="row" alignItems="center">
                  <FormControl>
                    <RadioGroupItem value="man" />
                  </FormControl>

                  <FormLabel>Man</FormLabel>
                </FormField>

                <FormField flexDirection="row" alignItems="center">
                  <FormControl>
                    <RadioGroupItem value="woman" />
                  </FormControl>
                  
                  <FormLabel>Woman</FormLabel>
                </FormField>
              </RadioGroup>
            </FormControl>

            <FormErrorMessage>{formState.errors.selected?.message}</FormErrorMessage>
          </FormField>
        )}
      />

      <Button type="submit">제출</Button>
    </FlexBox>
  )
}

export default Demo;`}
/>
