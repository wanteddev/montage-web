# ChipMultiSelect

두 개 이상의 옵션 선택(=다중 선택)이 필요한 경우에 사용합니다.

## 활용

- 단일 선택, 다중 선택 모두 가능해요.
- 선택 옵션이 한 개인 경우에는 쓰지 않아요.

### Multiple Select Form

<Demo code={`import { useId } from 'react';
import { Button, FlexBox, ChipMultiSelect, FormField, FormControl, FormLabel, FormErrorMessage } from '@wanteddev/wds';
import { useForm, Controller } from 'react-hook-form';

const Demo = () => {
  const id = useId();
  const form = useForm({
    defaultValues: {
      selected: [],
    },
  });

  const items = [
    {
      id: "develop",
      label: "개발",
    },
    {
      id: "design",
      label: "디자인",
    },
  ];

  return ( 
    <FlexBox
      as="form"
      flexDirection="column"
      gap="20px"
      onSubmit={form.handleSubmit((v) => alert(JSON.stringify(v)))}
    >
      <Controller
        control={form.control}
        name="selected"
        render={({ formState }) => (
          <FormField>
            <FormLabel>관심 분야</FormLabel>
            <FlexBox gap="8px">

              {items.map((item) => (
                <Controller
                  key={item.id}
                  control={form.control}
                  name="selected"
                  rules={{
                    required: {
                      value: true,
                      message: "필수 값입니다.",
                    },
                  }}
                  render={({ field, formState }) => {
                    return (
                      <FormField
                        key={item.id}
                        flexDirection="row"
                        alignItems="center"
                      >
                        <FormControl>
                          <ChipMultiSelect
                            {...field}
                            checked={field.value.includes(item.id)}
                            invalid={Boolean(formState.errors.selected)}
                            onCheckedChange={(checked) => {
                              return checked
                                ? field.onChange([...field.value, item.id])
                                : field.onChange(
                                    field.value.filter(
                                      (value) => value !== item.id
                                    )
                                  );
                            }}
                          >
                            {item.label}
                          </ChipMultiSelect>
                        </FormControl>
                      </FormField>
                    );
                  }}
                />
              ))}
            </FlexBox>

            <FormErrorMessage>{formState.errors.selected?.message}</FormErrorMessage>
          </FormField>
        )}
      />

      <Button type="submit">제출</Button>
    </FlexBox>
  )
}

export default Demo;`}
/>
