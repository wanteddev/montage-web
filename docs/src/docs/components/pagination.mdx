const Demo = () => {
  return (
    <Pagination count={10} />
  )
}

export default Demo;`}

### LeftContent, RightContent

variant가 `extended`일 때 콘텐츠를 추가할 수 있어요.

<Demo code={`import { Pagination, PaginationInput, PaginationSelect, FlexBox } from '@wanteddev/wds';

const Demo = () => {
  return (
    <FlexBox flexDirection="column" gap="40px">
      <FlexBox flexDirection="column" gap="24px">
        <Pagination 
          count={10}
          leftContent={<PaginationSelect />}
        />
        <Pagination 
          count={10}
          leftContent={<PaginationInput />}
        />
        <Pagination 
          count={10}
          leftContent={<span>커스텀</span>}
        />
      </FlexBox>

      <FlexBox flexDirection="column" gap="24px">
        <Pagination 
          count={10}
          rightContent={<PaginationSelect />} 
        />
        <Pagination 
          count={10}
          rightContent={<PaginationInput />} 
        />
        <Pagination 
          count={10}
          rightContent={<span>커스텀</span>} 
        />
      </FlexBox>

      <FlexBox flexDirection="column" gap="24px">
        <Pagination 
          count={10}
          leftContent={<PaginationSelect />}
          rightContent={<PaginationInput />} 
        />
        <Pagination 
          count={10} 
          leftContent={<span>커스텀</span>} 
          rightContent={<span>커스텀</span>} 
        />
      </FlexBox>
    </FlexBox>
  )
}

export default Demo;`}
/>
