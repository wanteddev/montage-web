# TextField

사용자가 특정한 텍스트를 입력하거나 편집할 수 있습니다.

## 활용

<Demo code={`import { TextField } from '@wanteddev/wds';

const Demo = () => {
  return (
    <TextField placeholder="입력하세요." width="300px" />
  )
}

export default Demo;`}
/>

## Example

### Action

<Demo code={`import { TextField, FlexBox, Label, TextButton } from '@wanteddev/wds';

const Demo = () => {

    <TextField
      placeholder="입력하세요."

### Form

<Demo code={`import { Button, FlexBox, TextField, Form, FormField, FormItem, FormControl, FormLabel, FormMessage } from '@wanteddev/wds';
import { useForm } from 'react-hook-form';

const Demo = () => {
  const form = useForm();

  return (
    <Form {...form}>
      <FlexBox
        as="form"
        flexDirection="column"
        gap="20px"
        onSubmit={form.handleSubmit((v) => alert(JSON.stringify(v)))}
      >
        <FormField
          control={form.control}
          name="content"
          rules={{
            required: {
              value: true,
              message: "필수 값입니다.",
            },
          }}
          render={({ field }) => (
            <FormItem>
              <FormLabel>내용</FormLabel>

              <FormControl>
                <TextField
                  {...field}
                  placeholder="입력하세요."
                  width="300px"
                />
              </FormControl>

              <FormMessage />
            </FormItem>
          )}
        />

        <Button type="submit">제출</Button>
      </FlexBox>
    </Form>
  )
}

export default Demo;`}
/>
