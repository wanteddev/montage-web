# ChipAction

동작을 설정하거나 실행합니다.(e.g. 화면 이동, 추가, 삭제)

## 활용

- Chip은 반응적이므로 콘텐츠의 특성에 따라 다른 동작을 제공해야 합니다.
- 텍스트에 말 줄임표(...) 사용을 지양합니다.
- 텍스트에 ‘#’을 붙이지 않아요.
- 클릭 또는 터치할 수 없는 요소에 사용하지 않아요.

}

### Form

<Demo code={`import { ChipAction, Button, FlexBox, FormField, FormLabel, FormControl, FormErrorMessage } from '@wanteddev/wds';
import { IconPlusThick } from '@wanteddev/wds-icon'
import { useForm, Controller } from 'react-hook-form';

const Demo = () => {
  const form = useForm({ defaultValues: { selected: [] } });

  const items = [
    {
      id: "develop",
      label: "개발",
    },
    {
      id: "design",
      label: "디자인",
    },
  ];

  return (
    <FlexBox
      as="form"
      flexDirection="column"
      gap="20px"
      onSubmit={form.handleSubmit((v) => alert(JSON.stringify(v)))}
    >
      <Controller
        control={form.control}
        name="selected"
        render={({ formState }) => (
          <FormField>
            <FormLabel required>관심 분야</FormLabel>

            <FlexBox gap="14px">
              {items.map((item) => (
                <Controller
                  key={item.id}
                  control={form.control}
                  name="selected"
                  rules={{
                    required: {
                      value: true,
                      message: "필수 값입니다.",
                    },
                  }}
                  render={({ field }) => {
                    const checked = field.value.includes(item.id);

                    return (
                      <FormField key={item.id}>
                        <FormControl>
                          <ChipAction
                            {...field}
                            active={checked}
                            onClick={() => {
                              return !checked
                                ? field.onChange([...field.value, item.id])
                                : field.onChange(
                                    field.value.filter(
                                      (value) => value !== item.id
                                    )
                                  );
                            }}
                          >
                            {item.label}
                          </ChipAction>
                        </FormControl>
                      </FormField>
                    );
                  }}
                />
              ))}
            </FlexBox>

            <FormErrorMessage>{formState.errors.selected?.message}</FormErrorMessage>
          </FormField>
        )}
      />

      <Button type="submit">적용</Button>
    </FlexBox>
  )
}

export default Demo;`}
/>
