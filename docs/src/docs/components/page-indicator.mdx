# PageIndicator

## 활용

<Demo code={`import { PageIndicator, Button, FlexBox } from '@wanteddev/wds';
import { useState } from 'react';

const Demo = () => {
  const [page, setPage] = useState(1);

  return (
    <FlexBox flexDirection="column" gap="24px">
      <FlexBox gap="10px">
        <Button variant="outlined" size="small" onClick={() => setPage((s) => s > 1 ? s - 1 : 1)}>Prev</Button>
        <Button variant="outlined" size="small" onClick={() => setPage((s) => s < 10 ? s + 1 : 10)}>Next</Button>
      </FlexBox>

      <PageIndicator onClickDot={(cur) => setPage(cur)} totalPage={10} currentPage={page} />
    </FlexBox>
  )
}

export default Demo;
`} />
