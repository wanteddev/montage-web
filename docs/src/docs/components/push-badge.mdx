# PushBadge

바뀐 상태나 알림을 표시할 때 사용합니다.

## 활용

<Demo code={`import { Avatar, PushBadge, FlexBox, IconButton, Switch } from '@wanteddev/wds';
import { IconBell } from '@wanteddev/wds-icon';
import { useState } from 'react';

const Demo = () => {
  const [invisible, setInvisible] = useState(true);

  return (
    <FlexBox flexDirection="column" gap="8px">
      <Switch checked={invisible} onCheckedChange={setInvisible} size="small" />

      <FlexBox gap="16px" alignItems="center">
        <IconButton size={24}>
          <PushBadge invisible={invisible}>
            <IconBell />
          </PushBadge>
        </IconButton>

        <PushBadge invisible={invisible}>
          <Avatar
            src="https://static.wanted.co.kr/oneid-user/2kcR2ULXRgEC7Rgk5psLTK/BwWtS7ZW8ryS2dpQrAW8Cr.png"
            alt="User"
          />
        </PushBadge>
      </FlexBox>
    </FlexBox>
  )
}

export default Demo;`}
/>
