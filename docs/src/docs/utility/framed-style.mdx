# FramedStyle

프레임 스타일을 관리합니다.

Sx Prop 혹은 `css`` ` 내부에서 사용 가능합니다.

## 활용

<Demo code={`import { Box, css, framedStyle, FlexBox } from '@wanteddev/wds';

const Demo = () => {
  return (
    <FlexBox flexDirection="column" gap={8} justifyContent="center">
      <FlexBox flexWrap="wrap" gap={8} alignItems="center">

## Props

<PropsTable component="framedStyle" fallback={[
{ name: 'invalid', types: 'boolean' },
{ name: 'disabled', types: 'boolean' },
{ name: 'selected', types: 'boolean' },
{ name: 'shadow', types: 'ThemeShadowToken' },
{ name: 'size', types: '\"small\" | \"medium\" | \"large\" | \"xlarge\" | undefined', defaultValue: 'medium' },
]} />


## Examples

### Selected

selected 속성을 사용하여 선택된 상태를 표현 할 수 있습니다.

<Demo code={`import { Box, framedStyle, FlexBox, SegmentedControl, SegmentedControlItem } from '@wanteddev/wds';
import { useState } from 'react';

const Demo = () => {
  const [status, setStatus] = useState('normal');

  const selected = status === 'selected';
  const invalid = status === 'negative';
  const disabled = status === 'disabled';

  return (
    <FlexBox flexDirection="column" gap={8}>
      <Box
        as="button"
        tabIndex={0}
        sx={framedStyle({
          size: 'small',
          selected,
          invalid,
          disabled
        })}
      >
        <Placeholder />
      </Box>

      <SegmentedControl
        value={status}
        onValueChange={setStatus}
      >
        <SegmentedControlItem value="normal">Normal</SegmentedControlItem>
        <SegmentedControlItem value="selected">Selected</SegmentedControlItem>
        <SegmentedControlItem value="negative">Negative</SegmentedControlItem>
        <SegmentedControlItem value="disabled">Disabled</SegmentedControlItem>
      </SegmentedControl>
    </FlexBox>
  )
}

const Placeholder = () => {
  return (
    <Box
      sx={theme => ({
        width: 100,
        height: 40,
        backgroundColor: theme.semantic.accent.background.violet,
        opacity: theme.opacity[8]
      })}
    />
  )
}


export default Demo;`}
/>
