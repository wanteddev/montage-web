import { Meta, ColorPalette, ColorItem } from '@storybook/blocks';

import { lightValues, darkValues } from '@wanteddev/wds';

<Meta title="Theme/Colors" />

# Color scheme

<br />
<br />

## Light Mode

<div>
  <ColorPalette>
    {Object.entries(lightValues).map(([k, v]) => {
      if (typeof v === 'string' && (v.match(/^#/) || v.match(/^rgb/))) {
        return <ColorItem key={k} title={k} colors={{ [k]: v }} />;
      } else if (typeof v === 'object') {
        return (
          <ColorItem
            key={k}
            title={k}
            colors={Object.entries(v).reduce((acc, [key, value]) => {
              if (
                typeof value === 'string' &&
                (value.match(/^#/) || value.match(/^rgb/))
              ) {
                return { ...acc, [key]: value };
              } else if (typeof value === 'object') {
                return { ...value };
              } else {
                return acc;
              }
            }, {})}
          />
        );
      }
      return null;
    })}
  </ColorPalette>
</div>
