### 자동으로 Navigation 높이 만큼 포지션 설정하기

<Markdown>
{`
\`\`\`tsx
import { useState } from 'react';
import { Alert, useSize, RegionConfig } from '@wanteddev/wds';

const Component = () => {
const [show, setShow] = useState(false);
const [nav, setNav] = useState<HTMLDivElement | null>(null);

const { height } = useSize(nav) || { height: 0 }

return (
<>
<TopNav ref={(node) => setNav(node)} />

      <Alert show={show} onShowChange={setShow} content="Test" />

      <RegionConfig viewportTop={height + 'px'} />

      <button onClick={() => setShow(true)} />
    </>

)
};
\`\`\`
`}
