## Custom

Responsive Props로는

`'size' | 'color' | 'accentColor' | 'variant' | 'css'` 사용 가능합니다.

<Markdown>
{`
\`\`\`tsx
import { css } from '@emotion/react';

import { useMedia, ContentBadge } from '@wanteddev/wds';

const Component = () => {
const { breakpoint, respondTo } = useMedia();

    return (
    	<ContentBadge
    		variant="outlined"
    		color="neutral"
    		size="xsmall"
    		lg={{ size: 'small' }}
    		css={{
    			[respondTo(breakpoint.md)]: {
    				marginTop: '5px',
    			},
    		}}
    	>
    		원티드
    	</ContentBadge>
    );

};
\`\`\`
`}
