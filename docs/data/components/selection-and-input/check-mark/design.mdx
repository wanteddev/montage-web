<SectionVariants
  components={['FormField', 'FlexBox', 'FormControl', 'FormLabel', 'CheckMark']}
  render={(value) => {
    const props = {
      tight: value['Tight'] === 'True',
      size: value['Size'] === 'Small' ? 'small' : 'medium',
    }

    const gap = props.tight ? '6px' : '4px'

    return `
      <FormField sx={{ width: '75%' }} flexDirection="row" gap="${gap}" alignItems="center" justifyContent="center">
        <FormControl>
          <CheckMark
            {...${JSON.stringify(props)}}
          />
        </FormControl>
        <FormLabel>Check mark</FormLabel>
      </FormField>
    `;
  }}
  variants={[
    {
      key: 'Size',
      options: [
        { label: 'Small', value: {} },
        { label: 'Medium', value: {} },
      ]
    },
    {
      key: 'Tight',
      options: [
        { label: 'False', value: {} },
        { label: 'True', value: {} },
      ],
    },
  ]}
/>
