<SectionVariants
  components={['FormField', 'FlexBox', 'FormControl', 'FormLabel', 'Checkbox']}
  render={(value) => {
    const props = {
      tight: value['Tight'] === 'True',
      size: value['Size'] === 'Small' ? 'small' : 'medium',
      bold: value['Bold'] === 'True',
    }

    const gap = props.tight ? '10px' : '8px'

    return `
      <FormField sx={{ width: '75%' }} flexDirection="row" gap="${gap}" alignItems="center" justifyContent="center">
        <FormControl>
          <Checkbox
            {...${JSON.stringify(props)}}
          />
        </FormControl>
        <FormLabel>Checkbox</FormLabel>
      </FormField>
    `;
  }}
  variants={[
    {
      key: 'Size',
      options: [
        { label: 'Small', value: {} },
        { label: 'Medium(Default)', value: {} },
      ]
    },
    {
      key: 'Tight',
      options: [
        { label: 'False', value: {} },
        { label: 'True', value: {} },
      ],
    },
    {
      key: 'Bold',
      options: [
        { label: 'True', value: {} },
        { label: 'False(Default)', value: {} },
      ],
    },
  ]}
/>
