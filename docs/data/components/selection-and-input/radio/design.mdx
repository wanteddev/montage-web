<SectionVariants
  components={['FormField', 'FlexBox', 'FormControl', 'FormLabel', 'RadioGroup', 'RadioGroupItem']}
  render={(value) => {
    const props = {
      tight: value['Tight'] === 'True',
      size: value['Size'] === 'Small' ? 'small' : 'medium',
      bold: value['Bold'] === 'True',
    }

    const gap = props.tight ? '10px' : '8px'

    return `
      <RadioGroup name="radio">
        <FormField flexDirection="row" gap="${gap}" alignItems="center">
          <FormControl>
            <RadioGroupItem
              value="radio"
              {...${JSON.stringify(props)}}
            />
          </FormControl>
          <FormLabel>Radio</FormLabel>
        </FormField>
      </RadioGroup>
    `;
  }}
  variants={[
    {
      key: 'Size',
      options: [
        { label: 'Small', value: {} },
        { label: 'Medium', value: {} },
      ]
    },
    {
      key: 'Tight',
      options: [
        { label: 'False', value: {} },
        { label: 'True', value: {} },
      ],
    },
    {
      key: 'Bold',
      options: [
        { label: 'True', value: {} },
        { label: 'False', value: {} },
      ],
    },
  ]}
/>
