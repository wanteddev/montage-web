## Basic popup

Popup overlay를 표시합니다.

breakpoint에 따라 variant를 override 하여 [Bottom sheet](/docs/components/presentation/bottom-sheet/web) 와 쉽게 통합할 수 있습니다.

<Demo code={`import {
  Button,
  Modal,
  ActionArea,
  ActionAreaButton,
  ModalContainer,
  ModalContent,
  ModalContentItem,
  ModalDescription,
  ModalHeading,
  ModalNavigation,
  ModalSummary,
  ModalTrigger,
} from '@montage-ui/core';

const Demo = () => {
  return (
    <Modal>
      <ModalTrigger>
        <Button>Open</Button>
      </ModalTrigger>

      <ModalContainer variant="popup">
        <ModalNavigation>
          Title
        </ModalNavigation>

        <ModalContent>
          <ModalContentItem>
            <ModalHeading>Heading</ModalHeading>
            <ModalSummary>Summary</ModalSummary>
            <ModalDescription>Description</ModalDescription>
          </ModalContentItem>
        </ModalContent>

        <ActionArea>
          <ActionAreaButton>
            Action
          </ActionAreaButton>
        </ActionArea>
      </ModalContainer>
    </Modal>
  )
}

export default Demo;`}
/>

## Anatomy

Popup 은 여러 컴포넌트를 조합해서 사용합니다.

`ModalTrigger` 는 선택적으로 사용할 수 있습니다.

기본 구성은 아래와 같습니다.

```tsx
<Modal>
  <ModalTrigger>
    <Button />
  </ModalTrigger>

  <ModalContainer>
    <ModalNavigation />

    <ModalContent>
      <ModalContentItem>
        <ModalHeading />
        <ModalSummary />
        <ModalDescription />
      </ModalContentItem>
    </ModalContent>

    <ActionArea>
      <ActionAreaButton />
    </ActionArea>
  </ModalContainer>
</Modal>
```

## Navigation

3가지 variant의 Navigation을 사용할 수 있습니다.
- emphasized (default)
- floating
- search

또한 `ModalNavigation` 의 leadingContent, trailingContent로 콘텐츠를 추가할 수 있으며 `ModalNavigationButton` 으로 사용합니다.

<Demo code={`import {
  FlexBox,
  Box,
  Button,
  Modal,
  ActionArea,
  ActionAreaButton,
  ModalContainer,
  ModalContent,
  ModalContentItem,
  ModalNavigation,
  ModalNavigationButton,
  ModalTrigger,
  SearchField,
  SegmentedControl,
  SegmentedControlItem,
  Typography,
} from '@montage-ui/core';
import { useState } from 'react';

const Control = ({ label, value, onValueChange, options }) => (
  <FlexBox flexDirection="column" gap="8px" sx={{ width: '100%', maxWidth: 400 }}>
    <Typography variant="label1" weight="medium" color="semantic.foreground.neutral.secondary">
      {label}
    </Typography>
    <SegmentedControl size="small" value={value} onValueChange={onValueChange}>
      {options.map((option) => (
        <SegmentedControlItem key={option} value={option}>
          {option}
        </SegmentedControlItem>
      ))}
    </SegmentedControl>
  </FlexBox>
);

const Demo = () => {
  const [navigationVariant, setNavigationVariant] = useState('emphasized');
  const [buttonVariant, setButtonVariant] = useState('icon-button');
  const [leading, setLeading] = useState('on');
  const [trailing, setTrailing] = useState('on');
  const [background, setBackground] = useState('off');

  // background only applies to icon buttons in a floating navigation
  const supportsBackground = navigationVariant === 'floating' && buttonVariant === 'icon-button';
  const iconButtonBackground = supportsBackground ? background === 'on' : undefined;

  const leadingContent = leading === 'on' ? (
    <ModalNavigationButton variant={buttonVariant === 'icon-button' ? 'back-button' : buttonVariant} background={iconButtonBackground}>
      {buttonVariant === 'text-button' ? '뒤로' : undefined}
    </ModalNavigationButton>
  ) : null;

  const trailingContent = trailing === 'on' ? (
    <ModalNavigationButton variant={buttonVariant === 'icon-button' ? 'close-button' : buttonVariant} background={iconButtonBackground}>
      {buttonVariant === 'text-button' ? '닫기' : undefined}
    </ModalNavigationButton>
  ) : null;

  const getNavigationContent = () => {
    switch (navigationVariant) {
      case 'search':
        return <SearchField placeholder="Enter search keyword." />;
      case 'floating':
        return null;
      case 'emphasized':
        return 'Emphasized';
    }
  };

  return (
    <FlexBox flexDirection="column" alignItems="center" gap="24px" sx={{ width: '100%' }}>
      <Control
        label="variant"
        value={navigationVariant}
        onValueChange={setNavigationVariant}
        options={['emphasized', 'floating', 'search']}
      />
      <Control
        label="button variant"
        value={buttonVariant}
        onValueChange={setButtonVariant}
        options={['icon-button', 'text-button']}
      />
      <Control label="leadingContent" value={leading} onValueChange={setLeading} options={['on', 'off']} />
      <Control label="trailingContent" value={trailing} onValueChange={setTrailing} options={['on', 'off']} />
      {supportsBackground && (
        <Control
          label="button background"
          value={background}
          onValueChange={setBackground}
          options={['on', 'off']}
        />
      )}

      <Modal>
        <ModalTrigger>
          <Button>Open</Button>
        </ModalTrigger>

        <ModalContainer variant="popup">
          <ModalNavigation
            variant={navigationVariant}
            leadingContent={leadingContent}
            trailingContent={trailingContent}
          >
            {getNavigationContent()}
          </ModalNavigation>

          <ModalContent>
            <ModalContentItem>
              <Box
                sx={theme => ({
                  width: '100%',
                  height: '800px',
                  background: theme.semantic.surface.accent.violetOpaque,
                  opacity: 0.08
                })}
              />
            </ModalContentItem>
          </ModalContent>

          <ActionArea>
            <ActionAreaButton>
              Action
            </ActionAreaButton>
          </ActionArea>
        </ModalContainer>
      </Modal>
    </FlexBox>
  )
}

export default Demo;`}
/>

## Content

내부 콘텐츠는 `ModalContent` 로 감싸서 사용하는 것을 권장하며, verticalPadding, horizontalPadding, gap 옵션을 사용하여 내부 여백을 조정할 수 있습니다.


`verticalPadding`
- none (default)
- top-only
- bottom-only
- both

`horizontalPadding`
- none
- both (default)

## With form

Form 구조로 사용하려면 `ModalContainer` 하위에 `<FlexBox as="form" flex="1" flexDirection="column">` 구조를 사용합니다.

<Demo code={`import {
  Button,
  Modal,
  ActionArea,
  ActionAreaButton,
  ModalContainer,
  ModalContent,
  ModalContentItem,
  ModalNavigation,
  ModalTrigger,
  TextField,
  FlexBox,
  FormControl,
  FormControlField,
  FormControlLabel,
  FormControlNegativeMessage,
  FormControlMessage,
} from '@montage-ui/core';
import { useForm, Controller } from 'react-hook-form';

const Demo = () => {
  const form = useForm({ defaultValues: { value: '' } });

  return (
    <Modal>
      <ModalTrigger>
        <Button>Open</Button>
      </ModalTrigger>

      <ModalContainer variant="popup">
        <FlexBox
          as="form"
          flex="1"
          flexDirection="column"
          onSubmit={form.handleSubmit((v) => alert(JSON.stringify(v)))}
        >
          <ModalNavigation>
            Title
          </ModalNavigation>

          <ModalContent>
            <ModalContentItem>
              <Controller
                control={form.control}
                name="value"
                rules={{
                  required: {
                    value: true,
                    message: '필수 값입니다.'
                  }
                }}
                render={({ field, formState }) => (
                  <FormControl>
                    <FormControlLabel>라벨</FormControlLabel>

                    <FormControlField>
                      <TextField {...field} width="100%" placeholder="값을 입력하세요." status={Boolean(formState.errors.value) ? 'negative' : 'normal'} />
                    </FormControlField>

                    {Boolean(formState.errors.value) ? (
                      <FormControlNegativeMessage>{formState.errors.value?.message}</FormControlNegativeMessage>
                    ): (
                      <FormControlMessage>Description</FormControlMessage>
                    )}
                  </FormControl>
                )} />
            </ModalContentItem>
          </ModalContent>

          <ActionArea>
            <ActionAreaButton type="submit">
              Submit
            </ActionAreaButton>
          </ActionArea>
        </FlexBox>
      </ModalContainer>
    </Modal>
  )
}

export default Demo;`}
/>
